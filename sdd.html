5.3 ResumeLookupTool (RAG)
pgvector HNSW 코사인 유사도 검색 · 목표 ≤300ms
class ResumeLookupTool(BaseTool):
  name = "resume_lookup"
  description = "이력서에서 관련 내용 검색"

  async def _arun(self, claim: str, session_id: str):
    # 1. 주장 텍스트 임베딩
    embedding = await openai_embed(claim)

    # 2. pgvector 코사인 유사도 Top-3
    chunks = await db.fetch("""
      SELECT content, metadata,
        1 - (embedding <=> $1) AS similarity
      FROM resume_chunks
      WHERE session_id = $2
      ORDER BY embedding <=> $1
      LIMIT 3
    """, embedding, session_id)

    # 3. 불일치 감지
    return RAGResult(
      chunks=chunks,
      discrepancy=detect_discrepancy(claim, chunks)
    )
5.4 WebFactCheckTool
Tavily Search API · 신뢰 도메인 필터 · 목표 ≤2.5s
class WebFactCheckTool(BaseTool):
  name = "web_fact_check"
  TRUSTED_DOMAINS = [
    "redis.io", "kubernetes.io",
    "docs.python.org", "aws.amazon.com",
    "github.com", "arxiv.org"
  ]

  async def _arun(self, claim: str, search_query: str):
    # 1. Tavily 검색 (신뢰 도메인 우선)
    results = await tavily.search(
      query=search_query,
      include_domains=self.TRUSTED_DOMAINS,
      max_results=5
    )

    # 2. GPT-4o 증거 분석
    verdict = await llm.ainvoke(
      FACT_JUDGE_PROMPT.format(
        claim=claim, evidence=results
      )
    )
    return WebResult(verdict=verdict, sources=results)

6.1 PII 마스킹 파이프라인
spaCy NER + 정규식 2단계 필터 · 원본은 암호화 후 별도 저장
class PIIMasker:
  def __init__(self):
    self.nlp = spacy.load("ko_core_news_lg")

  def mask(self, text: str) -> str:
    doc = self.nlp(text)
    masked = text

    # 1. NER 기반: PERSON, ORG, LOC
    for ent in reversed(doc.ents):
      if ent.label_ in ["PERSON","ORG","LOC"]:
        masked = masked[:ent.start_char] + \
                 f"[{ent.label_}_MASKED]" + \
                 masked[ent.end_char:]

    # 2. Regex: 전화번호, 이메일, 주민번호
    patterns = {
      "PHONE": r"01[016-9]-?\d{3,4}-?\d{4}",
      "EMAIL": r"[a-zA-Z0-9._%+-]+@[a-zA-Z0-9.-]+",
      "RRN":   r"\d{6}-[1-4]\d{6}",
    }
    for label, pat in patterns.items():
      masked = re.sub(pat, f"[{label}_MASKED]", masked)

    return masked

      # 입력은 해시만 저장 (개인정보 보호)
      "input_hash": sha256(input_text),
      "output_summary": sanitize(output),
      "latency_ms": latency_ms,
    })
